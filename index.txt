        // ฟังก์ชันคำนวณเวลาไทย (Asia/Bangkok) แบบเรียลไทม์
        function getThaiTimeInfo() {
            const now = new Date();
            const options = { timeZone: 'Asia/Bangkok', hour12: false, hour: '2-digit', minute: '2-digit', second: '2-digit' };
            const timeString = new Intl.DateTimeFormat('en-GB', options).format(now);
            const hour = parseInt(timeString.split(':')[0], 10);
            const isDaytime = hour >= 6 && hour < 19;
            return { timeString, hour, isDaytime };
        }

        // ระบบซิงค์ผู้เล่นออนไลน์เข้าสู่เซิร์ฟเวอร์กลาง (Shared Room Session)
        function syncOnlinePlayers(currentUser) {
            let onlineList = JSON.parse(localStorage.getItem('werewolf_online_room') || '[]');
            if (currentUser && !onlineList.some(p => p.username === currentUser.username)) {
                onlineList.push(currentUser);
                localStorage.setItem('werewolf_online_room', JSON.stringify(onlineList));
            }
            return onlineList;
        }

        // ระบบบันทึกข้อมูลอัตโนมัติ (Auto-Save Game State)
        function autoSaveGame() {
            if (!state.currentUser) return;
            const gameStateObj = {
                gameId: state.gameId || generateGameId(),
                dayNumber: state.dayNumber,
                gamePhase: state.gamePhase,
                players: state.players,
                currentUser: state.currentUser,
                lastNightSummary: state.lastNightSummary,
                timestamp: new Date().toISOString()
            };
            localStorage.setItem('werewolf_saved_session', JSON.stringify(gameStateObj));
        }

        // ระบบสมัคร/เข้าสู่ระบบด้วย Username และ PIN 4 หลัก
        function registerOrLoginUser(username, pin, isRegister = false) {
            return new Promise((resolve, reject) => {
                setTimeout(() => {
                    const serverDB = JSON.parse(localStorage.getItem('werewolf_server_users_db') || '{}');
                    const serverTimestamp = new Date().toISOString();

                    if (isRegister) {
                        if (serverDB[username]) {
                            return reject("ชื่อผู้ใช้งาน (Username / ID) นี้มีผู้ใช้งานแล้ว! กรุณาใช้ชื่ออื่น");
                        }
                        if (!pin || pin.length !== 4 || isNaN(pin)) {
                            return reject("รหัสผ่าน (PIN) ต้องเป็นตัวเลข 4 หลักเท่านั้น");
                        }
                        serverDB[username] = {
                            username: username,
                            pin: pin,
                            createdAt: serverTimestamp,
                            lastLogin: serverTimestamp,
                            loginCount: 1
                        };
                        localStorage.setItem('werewolf_server_users_db', JSON.stringify(serverDB));
                        resolve(serverDB[username]);
                    } else {
                        if (!serverDB[username]) {
                            return reject("ไม่พบชื่อผู้ใช้งานนี้ในระบบ กรุณาสร้างบัญชีก่อน");
                        }
                        if (serverDB[username].pin !== pin) {
                            return reject("รหัสผ่าน 4 หลักไม่ถูกต้อง!");
                        }
                        serverDB[username].lastLogin = serverTimestamp;
                        serverDB[username].loginCount = (serverDB[username].loginCount || 0) + 1;
                        localStorage.setItem('werewolf_server_users_db', JSON.stringify(serverDB));
                        resolve(serverDB[username]);
                    }
                }, 200);
            });
        }

        function generateGameId() {
            return 'WLF-' + Math.random().toString(36).substring(2, 8).toUpperCase();
        }

        function saveGameState() {
            const gameStateObj = {
                gameId: state.gameId || generateGameId(),
                dayNumber: state.dayNumber,
                gamePhase: state.gamePhase,
                players: state.players,
                currentUser: state.currentUser,

                timestamp: new Date().toISOString()
            };
            localStorage.setItem('werewolf_saved_session', JSON.stringify(gameStateObj));
            showModal("บันทึกสถานะเกม (Game ID: " + gameStateObj.gameId + ") เรียบร้อยแล้ว!");
        }

        function loadGameState() {
            const savedData = localStorage.getItem('werewolf_saved_session');
            if (!savedData) {
                showModal("ไม่พบข้อมูลเกมที่บันทึกไว้ในระบบ");
                return false;
            }
            try {
                const parsed = JSON.parse(savedData);
                state.gameId = parsed.gameId;
                state.dayNumber = parsed.dayNumber;
                state.gamePhase = parsed.gamePhase;
                state.players = parsed.players;
                state.currentUser = parsed.currentUser;
                state.lastNightSummary = parsed.lastNightSummary;

                if (state.currentUser) {
                    syncOnlinePlayers(state.currentUser);
                    if (state.gamePhase === 'waiting') {
                        state.view = 'waiting_room';
                    } else if (state.gamePhase === 'ended') {
                        state.view = 'summary_screen';
                    } else {
                        state.view = 'game_board';
                    }
                } else {
                    state.view = 'lobby_login';
                }
                render();
                showModal("โหลดข้อมูลเกม (Game ID: " + state.gameId + ") สำเร็จ!");
                return true;
            } catch (err) {
                showModal("เกิดข้อผิดพลาดในการโหลดข้อมูลเกม");
                return false;
            }
        }

                    allParticipants.push({
                        username: u.username,
                        status: u.status || 'อยู่ในห้องรอ',
                        isCurrent: state.currentUser && state.currentUser.username === u.username
                    });
                }
            });

        function toggleReady() {
            state.currentUser.isReady = !state.currentUser.isReady;
            state.currentUser.status = state.currentUser.isReady ? 'พร้อมแล้ว' : 'ยังไม่พร้อม';
            
            let onlineList = JSON.parse(localStorage.getItem('werewolf_online_room') || '[]');
            onlineList = onlineList.map(u => u.username === state.currentUser.username ? { ...u, status: state.currentUser.status, isReady: state.currentUser.isReady } : u);
            localStorage.setItem('werewolf_online_room', JSON.stringify(onlineList));

            checkAutoStart();
            render();
        }

        function simulateAddBot() {
            if (state.players.length >= 9) {
                showModal("ผู้เล่นเต็มจำนวน 10 คนแล้ว!");
                return;
            }
            const botId = state.players.length + 1;
            const botNames = ["เดชา", "วิชัย", "สมชาย", "มานพ", "ก้อง", "ธีร์", "ชัย", "ปูรณ์"];
            const randomName = botNames[state.players.length % botNames.length] + " (AI)";
            
            state.players.push({
                id: botId + 10,
                name: randomName,
                username: `bot_${botId}`,
                role: "villager",
                faction: "villager",
                status: "ยังมีชีวิต",
                isReady: true,
                isBot: true
            });
            checkAutoStart();
            render();
        }

        function checkAutoStart() {
            let onlineList = JSON.parse(localStorage.getItem('werewolf_online_room') || '[]');
            const allParticipants = [...onlineList, ...state.players];
            const allReady = allParticipants.every(p => p.isReady);
            if (allReady && allParticipants.length >= 6) {
                 startGameProcess();
            }
        }

        function forceStartGame() {
            let onlineList = JSON.parse(localStorage.getItem('werewolf_online_room') || '[]');
            const total = onlineList.length + state.players.length;
            if (total < 6) {
                showModal("ต้องการผู้เล่นอย่างน้อย 6 คนจึงจะเริ่มเกมได้ (ปัจจุบันมี " + total + " คน)");
                return;
            }
            startGameProcess();
        }

        function startGameProcess() {
            let onlineList = JSON.parse(localStorage.getItem('werewolf_online_room') || '[]');
            const allParticipants = [...onlineList, ...state.players];
            const total = allParticipants.length;

            let rolesPool = ['werewolf', 'werewolf', 'seer', 'doctor'];
            while (rolesPool.length < total) {
                rolesPool.push('villager');
            }
            rolesPool.sort(() => Math.random() - 0.5);

            state.currentUser.role = rolesPool[0];
            state.currentUser.status = 'กำลังเล่น';
            
            let assignedIndex = 1;
            state.players.forEach((p) => {
                p.role = rolesPool[assignedIndex++];
                p.status = 'ยังมีชีวิต';
            });

            state.gamePhase = 'night';
            state.dayNumber = 1;
            state.lastNightSummary = "เกมเริ่มต้นขึ้นแล้ว คืนแรกของหมู่บ้านอันเงียบสงบ...";
            state.view = 'role_screen';
            render();
        }

        }

        function submitNightAction(roleType) {
            const isAlive = state.currentUser.status === 'ยังมีชีวิต' || state.currentUser.status === 'กำลังเล่น' || state.currentUser.status === 'พร้อมแล้ว';
            if (!isAlive || state.currentUser.status === 'ออกจากเกมแล้ว' || state.currentUser.status === 'ตายแล้ว') {
                showModal("คุณออกจากเกมแล้ว ไม่สามารถใช้ความสามารถได้!");
                return;
            }

            const selectEl = document.getElementById('nightTargetSelect');
            if (!selectEl || !selectEl.value) {
                showModal("กรุณาเลือกเป้าหมายก่อนยืนยัน");
                return;
            }

            const targetVal = selectEl.value;
            if (roleType === 'werewolf') {
                state.nightActions.werewolfTarget = targetVal;
                showModal("คุณได้เลือกเป้าหมายการล่าในคืนนี้แล้ว");
            } else if (roleType === 'seer') {
                state.nightActions.seerTarget = targetVal;
                const targetPlayer = state.players.find(p => p.id == targetVal);
                const isWolf = targetPlayer && targetPlayer.role === 'werewolf';
                showModal(`ผลการตรวจสอบของผู้หยั่งรู้: เป้าหมาย "${targetPlayer ? (targetPlayer.name || targetPlayer.username) : 'ผู้เล่น'}" ${isWolf ? 'เป็นมนุษย์หมาป่า!' : 'ไม่ใช่มนุษย์หมาป่า'}`);
            } else if (roleType === 'doctor') {
                state.nightActions.doctorTarget = targetVal;
                showModal("คุณได้เลือกปกป้องผู้เล่นเป้าหมายในคืนนี้เรียบร้อยแล้ว");
            }

            setTimeout(() => {
                resolveNightPhase();
            }, 1000);
        }

        function resolveNightPhase() {
            const wolfTargetId = state.nightActions.werewolfTarget;
            const docTargetId = state.nightActions.doctorTarget;

            if (wolfTargetId) {
                if (wolfTargetId === docTargetId) {
                    state.lastNightSummary = "คืนที่ผ่านมา หมาป่าออกล่าเหยื่อ แต่หมอช่วยชีวิตไว้ทัน! ไม่มีผู้เสียชีวิต";
                } else {
                    const victim = state.players.find(p => p.id == wolfTargetId);
                    if (victim) {
                        victim.status = state.settings.deadText || 'ออกจากเกมแล้ว';
                        state.lastNightSummary = `คืนที่ผ่านมา น่าสลดใจ "${victim.name || victim.username}" ถูกมนุษย์หมาป่าจู่โจมและเสียชีวิต`;
                    }
                }
            } else {
                state.lastNightSummary = "คืนที่ผ่านมา สงบเรียบร้อย ไม่มีผู้เสียชีวิต";
            }

            state.gamePhase = 'day_discussion';
            if (checkWinCondition()) return;
            render();
        }

        function submitVoteAction() {
            const isAlive = state.currentUser.status === 'ยังมีชีวิต' || state.currentUser.status === 'กำลังเล่น' || state.currentUser.status === 'พร้อมแล้ว';
            if (!isAlive || state.currentUser.status === 'ออกจากเกมแล้ว' || state.currentUser.status === 'ตายแล้ว') {
                showModal("คุณออกจากเกมแล้ว ไม่สามารถลงคะแนนโหวตได้!");
                return;
            }

            const selectEl = document.getElementById('voteTargetSelect');
            if (!selectEl || !selectEl.value) {
                showModal("กรุณาเลือกผู้เล่นที่ต้องการโหวต");
                return;
            }

            const votedId = selectEl.value;
            const votedPlayer = state.players.find(p => p.id == votedId);
            
            if (votedPlayer) {
                votedPlayer.status = state.settings.deadText || 'ออกจากเกมแล้ว';
                state.lastNightSummary = `ผลการลงคะแนนประจำวัน: ชาวบ้านลงมติขับไล่ "${votedPlayer.name || votedPlayer.username}" ออกจากหมู่บ้านสำเร็จ`;
            }

            state.gamePhase = 'night';
            state.dayNumber++;

            if (checkWinCondition()) return;
            render();
        }

        function checkWinCondition() {
            const allAlive = [state.currentUser, ...state.players].filter(p => p.status === 'ยังมีชีวิต' || p.status === 'กำลังเล่น' || p.status === 'พร้อมแล้ว');
            const aliveWolves = allAlive.filter(p => p.role === 'werewolf');
            const aliveVillagers = allAlive.filter(p => p.role !== 'werewolf');

            if (aliveWolves.length === 0) {
                endGame("ฝ่ายชาวบ้านชนะ! กำจัดมนุษย์หมาป่าทั้งหมดได้สำเร็จ", "villager");
                return true;
            } else if (aliveWolves.length >= aliveVillagers.length) {
                endGame("ฝ่ายมนุษย์หมาป่าชนะ! จำนวนหมาป่าเท่ากับหรือมากกว่าชาวบ้าน", "werewolf");
                return true;
            }
            return false;
        }

        function endGame(summaryText, winningFaction) {
            state.lastNightSummary = summaryText;
            state.gamePhase = 'ended';

            const isUserWinner = (state.currentUser.role === 'werewolf' && winningFaction === 'werewolf') || 
                                 (state.currentUser.role !== 'werewolf' && winningFaction === 'villager');

            if (!state.scores[state.currentUser.username]) {
                state.scores[state.currentUser.username] = { name: state.currentUser.username, totalScore: 0, wins: 0, played: 0 };
            }
            state.scores[state.currentUser.username].played++;
            if (isUserWinner) {
                state.scores[state.currentUser.username].wins++;
                state.scores[state.currentUser.username].totalScore += state.settings.winPoints;
                state.currentUser.finalResult = "ชนะ";
            } else {
                state.scores[state.currentUser.username].totalScore += state.settings.lossPoints;
                state.currentUser.finalResult = "แพ้";
            }

            state.view = 'summary_screen';
            render();
        }

                                ${userArr.length === 0 ? '<div class="text-xs text-gray-500 text-center py-6">ยังไม่มีสมาชิกสมัครในระบบ</div>' : userArr.map((u, idx) => `
                                <div class="bg-slate-950/80 border border-slate-800 rounded-xl p-2.5 flex items-center justify-between text-xs">
                                    <div>
                                        <span class="font-bold text-amber-400">@${u.username}</span>
                                        <div class="text-[9px] text-gray-400">ล็อกอิน: ${u.loginCount || 1} ครั้ง</div>

                                <i class="fa-solid fa-shield-halved text-amber-400"></i>
                                <span>ตั้งค่ารหัสผ่านแอดมิน</span>
                            </h4>

                                <label class="block text-[11px] font-semibold text-gray-300 mb-1">รหัสผ่านปัจจุบัน</label>

                            <button onclick="updateAdminPassword()" class="w-full bg-slate-800 hover:bg-slate-700 text-amber-400 border border-slate-700 py-2 rounded-xl font-semibold transition">

        function updateAdminPassword() {
            const val = document.getElementById('adminNewPasswordInput').value.trim();
            if (!val) {
                showModal("รหัสผ่านผู้ดูแลต้องไม่ว่างเปล่า");
                return;
            }
            state.adminPassword = val;
            showModal("อัปเดตและบันทึกรหัสผ่านผู้ดูแลระบบเรียบร้อยแล้ว");
        }

        function resetAllUsers() {
            if (confirm("คุณแน่ใจหรือไม่ที่จะล้างข้อมูลบัญชีผู้ใช้ทั้งหมดในเซิร์ฟเวอร์?")) {
                localStorage.removeItem('werewolf_server_users_db');
                localStorage.removeItem('werewolf_online_room');
                showModal("ล้างข้อมูลสมาชิกและห้องออนไลน์ทั้งหมดเรียบร้อยแล้ว");
            }
        }

        function resetAllScores() {
            if (confirm("คุณแน่ใจหรือไม่ที่จะรีเซ็ตคะแนนสะสมทั้งหมด?")) {
                state.scores = {};
                showModal("รีเซ็ตคะแนนสะสมผู้เล่นทั้งหมดเรียบร้อยแล้ว");
            }
        }

        function goToLobby() {
            state.view = 'lobby_login';
            render();
        }

        function restartNewGame() {
            state.currentUser.status = 'อยู่ในห้องรอ';
            state.currentUser.isReady = false;
            state.currentUser.role = null;
            state.players.forEach(p => {
                p.status = 'ยังมีชีวิต';
                p.isReady = true;
            });
            state.view = 'waiting_room';
            render();
        }

        window.onload = function() {
            render();
            
            // อัปเดตหน้าจอและซิงค์รายชื่อผู้เล่นออนไลน์ทุกๆ 2 วินาที
            setInterval(() => {
                if (state.currentUser && (state.view === 'waiting_room' || state.view === 'game_board')) {
                    render();
                }
            }, 2000);

            // บันทึกข้อมูลเกมอัตโนมัติทุกๆ 10 วินาที
            setInterval(() => {
                if (state.currentUser) {
                    autoSaveGame();
                }
            }, 10000);
        };
    </script>
